cmake_minimum_required(VERSION 3.20)
project(CounterCraftNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This lab requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(COUNTERCRAFT_DEPS "" CACHE PATH "Pinned headers fetched by fetch-native-deps.ps1")
if(NOT EXISTS "${COUNTERCRAFT_DEPS}/json/json.hpp")
    message(FATAL_ERROR "Run scripts/fetch-native-deps.ps1 first")
endif()
add_library(countercraft_core STATIC src/frame.cpp src/receiver.cpp src/offline_guard.cpp)
target_include_directories(countercraft_core PUBLIC src)
target_include_directories(countercraft_core SYSTEM PUBLIC "${COUNTERCRAFT_DEPS}/json")
target_compile_definitions(countercraft_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(countercraft_core PUBLIC ws2_32)
add_library(countercraft_compositor STATIC src/compositor.cpp)
target_link_libraries(countercraft_compositor PUBLIC countercraft_core d3d11 d3dcompiler)
add_executable(countercraft_bench src/bench.cpp)
target_link_libraries(countercraft_bench PRIVATE countercraft_compositor)
add_executable(countercraft_protocol_tests tests/frame_tests.cpp)
target_link_libraries(countercraft_protocol_tests PRIVATE countercraft_core)
add_executable(countercraft_receiver_probe tests/receiver_probe.cpp)
target_link_libraries(countercraft_receiver_probe PRIVATE countercraft_core)
add_executable(countercraft_guard_tests tests/guard_tests.cpp)
target_link_libraries(countercraft_guard_tests PRIVATE countercraft_core)
if(NOT EXISTS "${COUNTERCRAFT_DEPS}/reshade/reshade.hpp")
    message(FATAL_ERROR "Missing pinned ReShade headers")
endif()
add_library(countercraft_addon SHARED src/addon.cpp)
set_target_properties(countercraft_addon PROPERTIES PREFIX "" SUFFIX ".addon64" OUTPUT_NAME "CounterCraftProbe")
target_include_directories(countercraft_addon SYSTEM PRIVATE "${COUNTERCRAFT_DEPS}/reshade")
target_link_libraries(countercraft_addon PRIVATE countercraft_core shell32)
add_dependencies(countercraft_guard_tests countercraft_addon)
foreach(target countercraft_core countercraft_compositor countercraft_bench countercraft_protocol_tests countercraft_receiver_probe countercraft_guard_tests countercraft_addon)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
enable_testing()
add_test(NAME native_protocol COMMAND countercraft_protocol_tests)
add_test(NAME native_gpu_depth COMMAND countercraft_bench)
add_test(NAME native_offline_guard COMMAND countercraft_guard_tests $<TARGET_FILE:countercraft_addon>)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME native_sockets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/socket_tests.py"
    --probe $<TARGET_FILE:countercraft_receiver_probe> -v)
set_tests_properties(native_sockets PROPERTIES TIMEOUT 30)
