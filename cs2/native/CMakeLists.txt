cmake_minimum_required(VERSION 3.20)
project(CounterCraftNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This lab requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(COUNTERCRAFT_DEPS "" CACHE PATH "Pinned headers fetched by fetch-native-deps.ps1")
if(NOT EXISTS "${COUNTERCRAFT_DEPS}/json/json.hpp")
    message(FATAL_ERROR "Run scripts/fetch-native-deps.ps1 first")
endif()
add_library(countercraft_core STATIC src/frame.cpp src/receiver.cpp src/offline_guard.cpp src/projection.cpp src/depth_inventory.cpp src/depth_observer.cpp src/host_camera.cpp)
target_include_directories(countercraft_core PUBLIC src)
target_include_directories(countercraft_core SYSTEM PUBLIC "${COUNTERCRAFT_DEPS}/json")
target_compile_definitions(countercraft_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(countercraft_core PUBLIC ws2_32)
add_library(countercraft_compositor STATIC src/compositor.cpp src/depth_sampler.cpp src/depth_readback.cpp src/camera_readback.cpp)
target_link_libraries(countercraft_compositor PUBLIC countercraft_core d3d11 d3dcompiler ole32)
add_executable(countercraft_bench src/bench.cpp)
target_link_libraries(countercraft_bench PRIVATE countercraft_compositor)
add_executable(countercraft_protocol_tests tests/frame_tests.cpp)
target_link_libraries(countercraft_protocol_tests PRIVATE countercraft_core)
add_executable(countercraft_receiver_probe tests/receiver_probe.cpp)
target_link_libraries(countercraft_receiver_probe PRIVATE countercraft_core)
add_executable(countercraft_guard_tests tests/guard_tests.cpp)
target_link_libraries(countercraft_guard_tests PRIVATE countercraft_core)
add_executable(countercraft_projection_tests tests/projection_tests.cpp)
target_link_libraries(countercraft_projection_tests PRIVATE countercraft_core)
add_executable(countercraft_depth_inventory_tests tests/depth_inventory_tests.cpp)
target_link_libraries(countercraft_depth_inventory_tests PRIVATE countercraft_core)
add_executable(countercraft_depth_observer_tests tests/depth_observer_tests.cpp)
target_link_libraries(countercraft_depth_observer_tests PRIVATE countercraft_core)
add_executable(countercraft_projection_gpu_tests tests/projection_gpu_tests.cpp)
target_link_libraries(countercraft_projection_gpu_tests PRIVATE countercraft_compositor)
add_executable(countercraft_depth_sampler_tests tests/depth_sampler_tests.cpp)
target_link_libraries(countercraft_depth_sampler_tests PRIVATE countercraft_compositor)
add_executable(countercraft_depth_readback_tests tests/depth_readback_tests.cpp)
target_link_libraries(countercraft_depth_readback_tests PRIVATE countercraft_compositor)
add_executable(countercraft_capture_control_tests tests/capture_control_tests.cpp)
add_executable(countercraft_host_camera_tests tests/host_camera_tests.cpp)
target_link_libraries(countercraft_host_camera_tests PRIVATE countercraft_core)
add_executable(countercraft_host_camera_probe tests/host_camera_probe.cpp)
target_link_libraries(countercraft_host_camera_probe PRIVATE countercraft_core)
add_executable(countercraft_camera_readback_tests tests/camera_readback_tests.cpp)
target_link_libraries(countercraft_camera_readback_tests PRIVATE countercraft_compositor)
target_compile_options(countercraft_camera_readback_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
target_compile_options(countercraft_host_camera_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
target_compile_options(countercraft_host_camera_probe PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(countercraft_capture_control_tests PRIVATE countercraft_core)
target_compile_options(countercraft_capture_control_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
if(NOT EXISTS "${COUNTERCRAFT_DEPS}/reshade/reshade.hpp")
    message(FATAL_ERROR "Missing pinned ReShade headers")
endif()
add_library(countercraft_addon SHARED src/addon.cpp src/mouse_input.cpp src/host_probe.cpp src/host_capture.cpp src/host_camera_feed.cpp)
target_include_directories(countercraft_depth_inventory_tests SYSTEM PRIVATE "${COUNTERCRAFT_DEPS}/reshade")
set_target_properties(countercraft_addon PROPERTIES PREFIX "" SUFFIX ".addon64" OUTPUT_NAME "CounterCraftProbe")
target_include_directories(countercraft_addon SYSTEM PRIVATE "${COUNTERCRAFT_DEPS}/reshade")
target_link_libraries(countercraft_addon PRIVATE countercraft_compositor shell32)
add_dependencies(countercraft_guard_tests countercraft_addon)
foreach(target countercraft_core countercraft_compositor countercraft_bench countercraft_protocol_tests countercraft_receiver_probe countercraft_guard_tests countercraft_addon countercraft_projection_tests countercraft_depth_inventory_tests countercraft_depth_observer_tests countercraft_projection_gpu_tests countercraft_depth_sampler_tests countercraft_depth_readback_tests)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
enable_testing()
add_executable(countercraft_client_viewport_tests tests/client_viewport_tests.cpp)
target_link_libraries(countercraft_client_viewport_tests PRIVATE countercraft_core)
add_test(NAME native_client_viewport COMMAND countercraft_client_viewport_tests)
add_executable(countercraft_mouse_tests tests/mouse_tests.cpp src/mouse_input.cpp)
target_link_libraries(countercraft_mouse_tests PRIVATE countercraft_core)
add_test(NAME native_mouse COMMAND countercraft_mouse_tests)
add_test(NAME native_protocol COMMAND countercraft_protocol_tests)
add_test(NAME native_gpu_depth COMMAND countercraft_bench)
add_test(NAME native_offline_guard COMMAND countercraft_guard_tests $<TARGET_FILE:countercraft_addon>)
add_test(NAME native_projection COMMAND countercraft_projection_tests)
add_test(NAME native_depth_inventory COMMAND countercraft_depth_inventory_tests)
add_test(NAME native_depth_observer COMMAND countercraft_depth_observer_tests)
add_test(NAME native_projection_gpu COMMAND countercraft_projection_gpu_tests)
add_test(NAME native_depth_sampler COMMAND countercraft_depth_sampler_tests)
add_test(NAME native_depth_readback COMMAND countercraft_depth_readback_tests)
add_test(NAME native_capture_control COMMAND countercraft_capture_control_tests)
add_test(NAME native_host_camera COMMAND countercraft_host_camera_tests)
add_test(NAME native_camera_readback COMMAND countercraft_camera_readback_tests)
set_tests_properties(native_camera_readback PROPERTIES TIMEOUT 30)
set_tests_properties(native_depth_sampler native_depth_readback PROPERTIES TIMEOUT 30)
set_tests_properties(native_projection native_depth_inventory native_depth_observer native_projection_gpu PROPERTIES TIMEOUT 60)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME native_sockets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/socket_tests.py"
    --probe $<TARGET_FILE:countercraft_receiver_probe> -v)
set_tests_properties(native_sockets PROPERTIES TIMEOUT 30)
